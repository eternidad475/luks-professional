  const CF_PRIMARY_RESULT_DEADLINE_MS=16000;
  async function cfFastPrimaryPreview(src,raw){
    if(!src||!src.dataUrl)return null;
    try{
      const img=await Promise.race([loadImage(src.dataUrl),new Promise((_,reject)=>setTimeout(()=>reject(new Error('quick preview decode timeout')),1400))]);
      const iw=img.naturalWidth||img.width,ih=img.naturalHeight||img.height;if(!iw||!ih)return src.dataUrl;
      const maxEdge=840,scale=Math.min(1,maxEdge/Math.max(iw,ih)),w=Math.max(1,Math.round(iw*scale)),h=Math.max(1,Math.round(ih*scale));
      const c=document.createElement('canvas');c.width=w;c.height=h;const x=c.getContext('2d',{alpha:false});x.imageSmoothingEnabled=true;x.imageSmoothingQuality='medium';x.drawImage(img,0,0,w,h);
      const facial=((src.category||'').toLowerCase()==='facial')||(!src.intraoral&&src.lips!==false);
      const rx=facial?w*.31:w*.12,ry=facial?h*.50:h*.18,rw=facial?w*.38:w*.76,rh=facial?h*.23:h*.55;
      const shade=((raw&&raw.shade)!=null?raw.shade:50),bright=1+Math.max(0,shade-50)/50*.06;
      x.save();x.beginPath();x.ellipse(rx+rw/2,ry+rh/2,rw/2,rh/2,0,0,Math.PI*2);x.clip();x.filter='brightness('+bright.toFixed(3)+') saturate(.93) contrast(1.02)';x.globalAlpha=.68;x.drawImage(img,0,0,w,h);x.restore();x.filter='none';
      try{drawBadge(x,w,h);}catch(e){}const out=c.toDataURL('image/jpeg',.80);c.width=c.height=1;return out;
    }catch(e){console.warn('[CaseFlow] quick preview uses source',e);return src.dataUrl;}
  }
  window.cfFastPrimaryPreview=cfFastPrimaryPreview;
  async function produceImage(src){
    const started=performance.now(),run={id:'primary_'+Date.now()+'_'+Math.random().toString(36).slice(2),primary:true,cancelled:false};
    let progressTimer=null,deadlineTimer=null;window.__cfPrimaryGenerationActive=true;window.__cfPrimaryRun=run;window.__caseflowQuickPreviewUsed=false;
    try{sessionStorage.setItem('cf_primary_generation_guard_v4',JSON.stringify({startedAt:Date.now(),id:run.id}));}catch(e){}
    if(typeof src.intraoral!=='boolean')src.intraoral=((src.category||'').toLowerCase()==='intraoral');
    if(typeof src.intraoralDark!=='boolean')src.intraoralDark=false;
    if(typeof src.lips!=='boolean')src.lips=(((src.category||'').toLowerCase()==='facial')&&!src.intraoral);
    try{if(window.cfWait)cfWait.show('AIシミュレーション中',0);}catch(e){}
    const quickTask=cfFastPrimaryPreview(src,state.simV96||{}),aiOn=aiEnabled();
    if(aiOn){const msgs=['画像を最適化中…','AIシミュレーション生成中…','臨床デザインを確認中…'];let mi=0;const tick=()=>{try{const q=document.querySelector('.cfWaitOverlay .cfWaitSub');if(q)q.textContent=msgs[mi++%msgs.length];}catch(e){}};tick();progressTimer=setInterval(tick,2200);}
    try{
      if(aiOn){
        const aiTask=generateViaAI(src,run).then(url=>({kind:'ai',url})).catch(err=>({kind:'error',err}));
        const deadlineTask=new Promise(resolve=>{deadlineTimer=setTimeout(()=>resolve({kind:'deadline'}),CF_PRIMARY_RESULT_DEADLINE_MS);});
        const result=await Promise.race([aiTask,deadlineTask]);
        if(result&&result.kind==='ai'&&result.url){window.__caseflowPrimaryResultMode='ai';return result.url;}
        if(result&&result.kind==='error'&&result.err&&result.err.cfNoTokens)throw result.err;
        run.cancelled=true;try{if(window.__cfGenerationAbort)window.__cfGenerationAbort();}catch(e){}
        if(result&&result.kind==='error')window.__caseflowLastAIError=String((result.err&&result.err.message)||result.err||'AI error');
      }
      let quick=null;try{quick=await Promise.race([quickTask,new Promise(resolve=>setTimeout(()=>resolve(src&&src.dataUrl||null),900))]);}catch(e){quick=src&&src.dataUrl||null;}
      if(!quick)throw new Error('一次プレビューを作成できませんでした');window.__caseflowQuickPreviewUsed=true;window.__caseflowPrimaryResultMode='quick';
      setTimeout(()=>{try{toastFn('通信が時間内に完了しなかったため、軽量プレビューを表示しました。結果画面から精密生成できます。');}catch(e){}},80);return quick;
    }finally{
      run.cancelled=true;clearInterval(progressTimer);clearTimeout(deadlineTimer);
      window.__caseflowGenerationTiming={total_ms:Math.round(performance.now()-started),mode:window.__caseflowPrimaryResultMode||'unknown',hard_deadline_ms:CF_PRIMARY_RESULT_DEADLINE_MS,completed_at:new Date().toISOString()};
      try{if(typeof window.cfLogEvent==='function')window.cfLogEvent('generation_timing','visual-simulation',window.__caseflowGenerationTiming);}catch(e){}
      try{sessionStorage.removeItem('cf_primary_generation_guard_v4');sessionStorage.removeItem('cf_primary_generation_guard_v3');}catch(e){}
      window.__cfPrimaryGenerationActive=false;if(window.__cfPrimaryRun===run)window.__cfPrimaryRun=null;try{if(window.cfWait)cfWait.hideAll();}catch(e){}
    }
  }
